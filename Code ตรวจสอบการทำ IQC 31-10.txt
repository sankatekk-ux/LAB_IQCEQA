/**
 * ฟังก์ชันดึงโครงสร้างรายการตรวจทั้งหมดจากชีต TestList และแปลงเป็น Object Map จัดกลุ่มตามสาขา
 */
function getTestListMap() {
  try {
    var spreadsheet = SpreadsheetApp.openById(PARENT_SHEET_ID);
    var sheet = spreadsheet.getSheetByName('TestList');

    if (!sheet) {
      return { status: "error", message: "ไม่พบแผ่นงานชื่อ 'TestList' ในระบบชีตหลัก" };
    }

    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return { status: "success", data: {} };
    }

    var headers = data[0];
    var sectionColIdx = headers.indexOf("สาขา");
    var testColIdx = headers.indexOf("รายการตรวจ");

    // ตั้งค่าคอลัมน์เริ่มต้นกรณีระบุชื่อหัวตารางไม่ตรง
    if (sectionColIdx === -1) sectionColIdx = 0;
    if (testColIdx === -1) testColIdx = 1;

    var map = {};
    for (var i = 1; i < data.length; i++) {
      var section = String(data[i][sectionColIdx]).trim();
      var test = String(data[i][testColIdx]).trim();

      if (section && test) {
        if (!map[section]) {
          map[section] = [];
        }
        // ตรวจสอบข้อมูลซ้ำก่อนเพิ่มลงในอาร์เรย์
        if (map[section].indexOf(test) === -1) {
          map[section].push(test);
        }
      }
    }
    return { status: "success", data: map };
  } catch (error) {
    return { status: "error", message: "เกิดข้อผิดพลาดในการดึงรายการตรวจ: " + error.toString() };
  }
}

/**
 * ฟังก์ชันค้นหาชื่อผู้ปฏิบัติงานจากแผ่นงาน Personel
 * โดยเทียบรหัสผู้ปฏิบัติ (2 ตัวท้าย) กับ "เลขที่ใบประกอบวิชาชีพ" (2 ตัวท้าย)
 * ใช้ร่วมกันทั้งหน้า IQC (operatorId) และหน้า EQA (reviewerId / headId)
 * @param {string} operatorId รหัสที่ผู้ใช้ป้อนเข้ามา (เลข 2 หลัก)
 */
function findOperatorName(operatorId) {
  try {
    var spreadsheet = SpreadsheetApp.openById(PARENT_SHEET_ID);
    var sheet = spreadsheet.getSheetByName('Personel');

    if (!sheet) {
      return { status: "error", message: "ไม่พบแผ่นงานชื่อ 'Personel' ในระบบ กรุณาสร้างแผ่นงานนี้ในชีตหลัก" };
    }

    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return { status: "error", message: "ไม่พบข้อมูลรายชื่อบุคลากรในแผ่นงาน Personel" };
    }

    var headers = data[0];
    var licenseColIdx = headers.indexOf("เลขที่ใบประกอบวิชาชีพ");
    var nameColIdx = headers.indexOf("ชื่อ");

    // หากไม่พบคอลัมน์ตามชื่อที่ระบุ ให้ใช้ค่า Default (A=ใบประกอบวิชาชีพ, B=ชื่อ)
    if (licenseColIdx === -1) licenseColIdx = 0;
    if (nameColIdx === -1) nameColIdx = 1;

    var searchId = String(operatorId).trim();

    for (var i = 1; i < data.length; i++) {
      var licenseNum = String(data[i][licenseColIdx]).trim();
      var fullName = String(data[i][nameColIdx]).trim();

      if (licenseNum.length >= 2) {
        // ดึงเฉพาะเลข 2 ตัวท้ายของใบประกอบวิชาชีพมาเปรียบเทียบ
        var lastTwoDigits = licenseNum.substring(licenseNum.length - 2);
        if (lastTwoDigits === searchId) {
          return { status: "success", name: fullName };
        }
      }
    }

    return { status: "not_found", message: "ไม่พบข้อมูลผู้ปฏิบัติงานรหัสนี้ในระบบชีต" };
  } catch (error) {
    return { status: "error", message: "เกิดข้อผิดพลาดในการเชื่อมต่อฐานข้อมูล: " + error.toString() };
  }
}

/**
 * ฟังก์ชันดึงข้อมูลสรุปผล IQC ทั้งหมดมาแสดงใน Dashboard
 * โดยเชื่อมโยง IQC_Child เข้ากับ IQC_Parent และเรียงลำดับวันที่จากล่าสุดลงมา
 */
function getIqcDashboardData() {
  try {
    var parentSheet = SpreadsheetApp.openById(PARENT_SHEET_ID).getSheetByName('IQC_Parent');
    var childSheet = SpreadsheetApp.openById(CHILD_SHEET_ID).getSheetByName('IQC_Child');

    if (!parentSheet || !childSheet) {
      return { status: "error", message: "ไม่พบแผ่นงาน IQC_Parent หรือ IQC_Child ใน Google Sheets" };
    }

    var parentData = parentSheet.getDataRange().getValues();
    var childData = childSheet.getDataRange().getValues();

    if (parentData.length <= 1 || childData.length <= 1) {
      return { status: "success", data: [] };
    }

    // แปลงข้อมูล Parent เป็น Map โดยมี parentId เป็น Key
    // Headers ของ Parent: [operatorId, operatorName, parentId, recordDate, shift, section, machine]
    var parentMap = {};
    for (var i = 1; i < parentData.length; i++) {
      var row = parentData[i];
      var pId = String(row[2]).trim();
      if (pId) {
        var rawDate = row[3];
        var formattedDate = "";
        var timeVal = 0;

        if (rawDate instanceof Date) {
          formattedDate = Utilities.formatDate(rawDate, "GMT+7", "yyyy-MM-dd");
          timeVal = rawDate.getTime();
        } else if (rawDate) {
          formattedDate = String(rawDate).trim();
          var parsedDate = new Date(rawDate);
          timeVal = isNaN(parsedDate.getTime()) ? 0 : parsedDate.getTime();
        }

        parentMap[pId] = {
          operatorId: String(row[0] || ""),
          operatorName: String(row[1] || ""),
          recordDate: formattedDate,
          timeVal: timeVal,
          shift: String(row[4] || ""),
          section: String(row[5] || ""),
          machine: String(row[6] || "")
        };
      }
    }

    // Headers ของ Child: [childId, parentId, testItem, westgard, numControl, errorType, reagentStatus, controlStatus, standardStatus, instrumentStatus, rootCause, correction, result]
    var resultList = [];
    for (var j = 1; j < childData.length; j++) {
      var cRow = childData[j];
      var pId = String(cRow[1]).trim();
      var pInfo = parentMap[pId] || { recordDate: "-", section: "-", machine: "-", timeVal: 0 };

      resultList.push({
        childId: String(cRow[0] || ""),
        parentId: pId,
        recordDate: pInfo.recordDate,
        timeVal: pInfo.timeVal,
        section: pInfo.section,
        machine: pInfo.machine,
        operatorName: pInfo.operatorName || "",
        testItem: String(cRow[2] || ""),
        westgard: String(cRow[3] || ""),
        numControl: String(cRow[4] || ""),
        errorType: String(cRow[5] || ""),
        reagentStatus: String(cRow[6] || ""),
        controlStatus: String(cRow[7] || ""),
        standardStatus: String(cRow[8] || ""),
        instrumentStatus: String(cRow[9] || ""),
        rootCause: String(cRow[10] || ""),
        correction: String(cRow[11] || ""),
        result: String(cRow[12] || "")
      });
    }

    // เรียงลำดับตามเวลาจากล่าสุดขึ้นบนสุด (Descending Order)
    resultList.sort(function(a, b) {
      return b.timeVal - a.timeVal;
    });

    return { status: "success", data: resultList };

  } catch (error) {
    return { status: "error", message: "เกิดข้อผิดพลาดในการดึงข้อมูล Dashboard: " + error.toString() };
  }
}

/**

 */

    var childSheet = SpreadsheetApp.openById(CHILD_SHEET_ID).getSheetByName('IQC_Child');

      formObj.parent.operatorName,
      parentId,
      formObj.parent.recordDate,
      formObj.parent.shift,

     ]);

      var item = formObj.children[i];

      childDataRows.push([
        childId,

        item.westgard,

        item.errorType,

        item.controlStatus,

/**
 * ฟังก์ชันบันทึกข้อมูลแบบฟอร์ม EQA (ทั้ง Parent และ Children ที่ไม่ผ่านเกณฑ์)
 * ต้องมีแผ่นงาน EQA_Parent และ EQA_Child ในสเปรดชีตเดียวกัน
 * EQA_Parent columns:
 *   parentId | reviewDate | reviewerId | reviewerName | program | trialNumber |
 *   testDate | totalItems | passedItems | headId | headName | headAckDate
 * EQA_Child columns:
 *   childId | parentId | testNopass | trendHistory | docProblem | prepProblem |
 *   iqcReview | trendMatch | dilution | probAmount | impact | conclusion | correction
 */
function saveEqaData(formObj) {
  try {
    var spreadsheet = SpreadsheetApp.openById(PARENT_SHEET_ID);
    var parentSheet = spreadsheet.getSheetByName('EQA_Parent');
    var childSheet = spreadsheet.getSheetByName('EQA_Child');

    if (!parentSheet) throw new Error("ไม่พบแผ่นงานชื่อ EQA_Parent กรุณาสร้างแผ่นงานนี้ในสเปรดชีตหลักก่อน");
    if (!childSheet) throw new Error("ไม่พบแผ่นงานชื่อ EQA_Child กรุณาสร้างแผ่นงานนี้ในสเปรดชีตหลักก่อน");

    var p = formObj.parent;
    var totalItemsCheck = Number(p.totalItems) || 0;
    var passedItemsCheck = Number(p.passedItems) || 0;
    var isAllPassed = totalItemsCheck > 0 && passedItemsCheck >= totalItemsCheck;

    // อนุญาตให้ไม่มีรายการ Child (Nopass) ได้เฉพาะกรณีผ่านครบ 100% เท่านั้น
    // ถ้ามีรายการตกเกณฑ์จริง (passedItems < totalItems) ต้องมีอย่างน้อย 1 รายการเสมอ
    if (!isAllPassed && (!formObj.children || formObj.children.length === 0)) {
      throw new Error("ไม่พบรายการ EQA ที่ไม่ผ่านเกณฑ์ กรุณาเพิ่มรายการอย่างน้อย 1 รายการ");
    }

    var dateObj = new Date();
    var timestamp = Utilities.formatDate(dateObj, "GMT+7", "yyyyMMdd-HHmmss");
    var parentId = "EQA-" + timestamp;
    var attachment = formObj.attachment || {};

    parentSheet.appendRow([
      parentId,
      p.reviewDate,
      p.reviewerId,
      p.reviewerName,
      p.program,
      p.trialNumber,
      p.testDate,
      p.totalItems,
      p.passedItems,
      p.headId ? p.headId : "-",           // หากไม่มีข้อมูลให้ใส่เป็นค่าว่าง
      p.headName ? p.headName : "-",       // หากไม่มีข้อมูลให้ใส่เป็นค่าว่าง
      p.headAckDate ? p.headAckDate : "-", // หากไม่มีข้อมูลให้ใส่เป็นค่าว่าง
      attachment.fileId || '',
      attachment.fileName || '',
      attachment.fileUrl || ''
    ]);

    var childDataRows = [];
    for (var i = 0; i < formObj.children.length; i++) {
      var item = formObj.children[i];
      var childId = parentId + "-C" + (i + 1);

      childDataRows.push([
        childId,
        parentId,
        item.testNopass,
        item.trendHistory,
        item.docProblem,
        item.prepProblem,
        item.iqcReview,
        item.trendMatch,
        item.dilution,
        item.probAmount,
        item.impact,
        item.conclusion,
        item.correction
      ]);
    }

    if (childDataRows.length > 0) {
      childSheet.getRange(childSheet.getLastRow() + 1, 1, childDataRows.length, childDataRows[0].length).setValues(childDataRows);
    }

    return { status: "success", message: "บันทึกข้อมูล EQA เรียบร้อยแล้ว", parentId: parentId };

  } catch (error) {
    return { status: "error", message: error.toString() };
  }
}
  /**
 * ฟังก์ชันดึงข้อมูลสรุปผล EQA ทั้งหมดมาแสดงใน Dashboard EQA
 */
function getEqaDashboardData() {
  try {
    var ss = SpreadsheetApp.openById(PARENT_SHEET_ID);
    var parentSheet = ss.getSheetByName('EQA_Parent');
    var childSheet  = ss.getSheetByName('EQA_Child');

    if (!parentSheet || !childSheet) {
      return { status: "error", message: "ไม่พบแผ่นงาน EQA_Parent หรือ EQA_Child ใน Google Sheets" };
    }

    var parentData = parentSheet.getDataRange().getValues();
    var childData  = childSheet.getDataRange().getValues();

    if (parentData.length <= 1) return { status: "success", data: [] };

    // 1) จัดกลุ่ม Child ตาม parentId
    var childMap = {};
    for (var j = 1; j < childData.length; j++) {
      var cRow = childData[j];
      var cpId = String(cRow[1] || "").trim();
      if (!cpId) continue;
      if (!childMap[cpId]) childMap[cpId] = [];
      childMap[cpId].push(cRow);
    }

    // 2) วนจาก Parent เป็นหลัก -> ครั้งที่ผ่านหมดก็จะติดมาด้วย
    var resultList = [];
    var usedParent = {};

    for (var i = 1; i < parentData.length; i++) {
      var row = parentData[i];
      var pId = String(row[0] || "").trim();
      if (!pId) continue;
      usedParent[pId] = true;

      var rawDate = row[1], formattedDate = "", timeVal = 0;
      if (rawDate instanceof Date) {
        formattedDate = Utilities.formatDate(rawDate, "GMT+7", "yyyy-MM-dd");
        timeVal = rawDate.getTime();
      } else if (rawDate) {
        formattedDate = String(rawDate).trim();
        var pd = new Date(rawDate);
        timeVal = isNaN(pd.getTime()) ? 0 : pd.getTime();
      }

      var total  = Number(row[7]) || 0;
      var passed = Number(row[8]) || 0;

      var base = {
        parentId: pId,
        reviewDate: formattedDate,
        timeVal: timeVal,
        reviewerName: String(row[3] || ""),
        program: String(row[4] || ""),
        trialNumber: String(row[5] || ""),
        totalItems: total,
        passedItems: passed
      };

      var kids = childMap[pId] || [];

      if (kids.length === 0) {
        // ครั้งที่ผ่านทุกรายการ
        resultList.push(Object.assign({}, base, {
          childId: "",
          isFail: false,
          status: "ผ่านทุกรายการ",
          testNopass: "-", iqcReview: "-", probAmount: "-",
          impact: "-", conclusion: "-", correction: "-"
        }));
      } else {
        kids.forEach(function (cRow) {
          resultList.push(Object.assign({}, base, {
            childId: String(cRow[0] || ""),
            isFail: true,
            status: "ไม่ผ่าน",
            testNopass: String(cRow[2] || ""),
            iqcReview:  String(cRow[6] || ""),
            probAmount: String(cRow[9] || ""),
            impact:     String(cRow[10] || ""),
            conclusion: String(cRow[11] || ""),
            correction: String(cRow[12] || "")
          }));
        });
      }
    }

    // 3) กันข้อมูลกำพร้า (Child ที่หา Parent ไม่เจอ) ไม่ให้หายไปเงียบ ๆ
    Object.keys(childMap).forEach(function (orphanId) {
      if (usedParent[orphanId]) return;
      childMap[orphanId].forEach(function (cRow) {
        resultList.push({
          childId: String(cRow[0] || ""), parentId: orphanId,
          reviewDate: "-", timeVal: 0, program: "-", reviewerName: "",
          trialNumber: "", totalItems: 0, passedItems: 0,
          isFail: true, status: "ไม่ผ่าน",
          testNopass: String(cRow[2] || ""), iqcReview: String(cRow[6] || ""),
          probAmount: String(cRow[9] || ""), impact: String(cRow[10] || ""),
          conclusion: String(cRow[11] || ""), correction: String(cRow[12] || "")
        });
      });
    });

    resultList.sort(function (a, b) { return b.timeVal - a.timeVal; });
    return { status: "success", data: resultList };

  } catch (error) {
    return { status: "error", message: "เกิดข้อผิดพลาดในการดึงข้อมูล EQA Dashboard: " + error.toString() };
  }
}

function getEqaPassRateSummary(startDate, endDate) {
  try {
    var sheet = SpreadsheetApp.openById(PARENT_SHEET_ID).getSheetByName('EQA_Parent');
    if (!sheet) throw new Error('ไม่พบแผ่นงาน EQA_Parent');

    var rows = sheet.getDataRange().getValues();
    var start = startDate ? new Date(startDate + 'T00:00:00') : null;
    var end = endDate ? new Date(endDate + 'T23:59:59') : null;

    var totalAll = 0;
    var passedAll = 0;
    var programMap = {};

    for (var i = 1; i < rows.length; i++) {
      var row = rows[i];
      var reviewDate = row[1];
      var program = String(row[4] || 'ไม่ระบุ Program').trim();
      var totalItems = Number(row[7]) || 0;
      var passedItems = Number(row[8]) || 0;

      if (reviewDate instanceof Date) {
        if (start && reviewDate < start) continue;
        if (end && reviewDate > end) continue;
      }

      if (totalItems <= 0) continue;

      totalAll += totalItems;
      passedAll += passedItems;

      if (!programMap[program]) {
        programMap[program] = {
          program: program,
          totalItems: 0,
          passedItems: 0
        };
      }

      programMap[program].totalItems += totalItems;
      programMap[program].passedItems += passedItems;
    }

    var byProgram = Object.keys(programMap).map(function(program) {
      var item = programMap[program];
      item.passRate = Number(
        ((item.passedItems / item.totalItems) * 100).toFixed(2)
      );
      return item;
    }).sort(function(a, b) {
      return b.passRate - a.passRate;
    });

    return {
      status: 'success',
      overall: {
        totalItems: totalAll,
        passedItems: passedAll,
        passRate: totalAll
          ? Number(((passedAll / totalAll) * 100).toFixed(2))
          : 0
      },
      byProgram: byProgram
    };

  } catch (error) {
    return { status: 'error', message: error.toString() };
  }
}



/** ===================== ค้นหา / แก้ไข IQC + EQA ===================== */
function _formatDateForClient_(value) {
  if (value instanceof Date) return Utilities.formatDate(value, 'GMT+7', 'yyyy-MM-dd');
  if (value === null || value === undefined || value === '') return '';
  var d = new Date(value);
  return isNaN(d.getTime()) ? String(value) : Utilities.formatDate(d, 'GMT+7', 'yyyy-MM-dd');
}

function _sheetRowsByHeaders_(sheet) {
  var values = sheet.getDataRange().getValues();
  if (!values.length) return {headers: [], rows: []};
  return {headers: values[0].map(function(h){ return String(h).trim(); }), rows: values.slice(1)};
}

function _findHeader_(headers, names, fallback) {
  for (var i = 0; i < names.length; i++) {
    var idx = headers.indexOf(names[i]);
    if (idx !== -1) return idx;
  }
  return fallback;
}

function _deleteChildRowsByParentId_(sheet, parentId, parentIdColIndex) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return;
  var values = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getValues();
  for (var i = values.length - 1; i >= 0; i--) {
    if (String(values[i][parentIdColIndex] || '').trim() === String(parentId).trim()) {
      sheet.deleteRow(i + 2);
    }
  }
}

    }

    var bytes = Utilities.base64Decode(fileObj.base64);
    if (bytes.length > EQA_MAX_FILE_SIZE) {
      throw new Error('ไฟล์มีขนาดเกิน 5 MB');
    }

    }

    var blob = Utilities.newBlob(bytes, fileObj.mimeType, safeName);
    var file = DriveApp.getFolderById(EQA_DRIVE_FOLDER_ID).createFile(blob);
 
    return {
      status: 'success',
      fileId: file.getId(),
      fileName: file.getName(),
      fileUrl: file.getUrl()
    };

function _eqaPdfDate_(value, tz) {
  if (!value) return '';
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value.getTime())) return Utilities.formatDate(value,tz,'dd/MM/yyyy');
  var d = new Date(value);
  if (!isNaN(d.getTime())) return Utilities.formatDate(d,tz,'dd/MM/yyyy');
  return String(value);
}

function normalizeToISO_(v, tz) {
  var y, m, d;
  if (v instanceof Date) {
    if (isNaN(v.getTime())) return null;
    var s = Utilities.formatDate(v, tz, 'yyyy-M-d').split('-');
    y = Number(s[0]); m = Number(s[1]); d = Number(s[2]);
  } else if (typeof v === 'string') {
    var mt = v.trim().match(/(\d{1,4})[\/\-.](\d{1,2})[\/\-.](\d{1,4})/);
    if (!mt) return null;
    if (mt[1].length === 4) { y = Number(mt[1]); m = Number(mt[2]); d = Number(mt[3]); }
    else { d = Number(mt[1]); m = Number(mt[2]); y = Number(mt[3]); }
  } else {
    return null;
  }
  if (y > 2400) y -= 543;   // พ.ศ. -> ค.ศ. (ใช้กับ Date ด้วย)
  if (y < 100) y += 2000;
  return y + '-' + String(m).padStart(2, '0') + '-' + String(d).padStart(2, '0');
}
