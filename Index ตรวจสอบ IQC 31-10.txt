        // แปลงวันที่ของแถวเป็น timestamp (เที่ยงคืนตามวัน)
        function getEqaItemTime(item) {
          if (item.timeVal && item.timeVal > 0) {
              const d = new Date(item.timeVal);
              return new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime();
          }
          const d2 = new Date(item.reviewDate);
          return isNaN(d2.getTime()) ? null : new Date(d2.getFullYear(), d2.getMonth(), d2.getDate()).getTime();
        }

        function filterEqaDashboardTable() {
            const programFilter = document.getElementById('eqaDashProgramFilter').value;
            const searchText    = document.getElementById('eqaDashSearchInput').value.toLowerCase().trim();
            const startVal      = (document.getElementById('eqaDashStartDate') || {}).value || '';
            const endVal        = (document.getElementById('eqaDashEndDate')   || {}).value || '';
            const onlyFail      = (document.getElementById('eqaDashOnlyFail') || {}).checked || false;

            const startTime = startVal ? new Date(startVal + 'T00:00:00').getTime() : null;
            const endTime   = endVal   ? new Date(endVal   + 'T00:00:00').getTime() : null;

            const filtered = rawEqaDashboardData.filter(item => {
                const matchProgram = !programFilter || item.program === programFilter;
                const matchSearch  = !searchText || Object.values(item).some(v => String(v).toLowerCase().includes(searchText));
                const matchFail    = !onlyFail || item.isFail === true;

                let matchDate = true;
                if (startTime !== null || endTime !== null) {
                    const t = getEqaItemTime(item);
                    if (t === null) matchDate = false;                       // ไม่มีวันที่ = ตัดออกเมื่อกรองวันที่
                    else {
                        if (startTime !== null && t < startTime) matchDate = false;
                        if (endTime   !== null && t > endTime)   matchDate = false;
                    }
                }
                return matchProgram && matchSearch && matchDate && matchFail;
            });

            renderEqaDashboardTable(filtered);
            // กราฟ/สถิติ ให้นับเฉพาะรายการที่ไม่ผ่านเท่านั้น
            updateEqaDashboardAnalytics(filtered.filter(i => i.isFail));
        }

        function updateDashboardAnalytics(data) {
            const countEl = document.getElementById('kpiTotalCount');
            if(countEl) countEl.innerText = data.length;

            const sectionCounts = {};
            const machineCounts = {};
            const testItemCounts = {};
            const causeCounts = {};
            const correctionCounts = {};

            data.forEach(item => {
                if(item.section) sectionCounts[item.section] = (sectionCounts[item.section] || 0) + 1;
                if(item.machine) machineCounts[item.machine] = (machineCounts[item.machine] || 0) + 1;
                if(item.testItem) testItemCounts[item.testItem] = (testItemCounts[item.testItem] || 0) + 1;

                if (item.rootCause) {
                    item.rootCause.split(',').forEach(c => {
                        const cause = c.trim();
                        if (cause) causeCounts[cause] = (causeCounts[cause] || 0) + 1;
                    });
                }

                if (item.correction) {
                    item.correction.split(',').forEach(c => {
                        const corr = c.trim();
                        if (corr) correctionCounts[corr] = (correctionCounts[corr] || 0) + 1;
                    });
                }
            });

            const getTopN = (obj, n) => Object.entries(obj).sort((a,b) => b[1] - a[1]).slice(0, n);

            const sortedCauses = getTopN(causeCounts, 8);
            const sortedCorrections = getTopN(correctionCounts, 8);
            const sortedTestItems = getTopN(testItemCounts, 10);

            const causeEl = document.getElementById('kpiTopCause');
            const corrEl = document.getElementById('kpiTopCorrection');
            if(causeEl) causeEl.innerText = sortedCauses.length > 0 ? `${sortedCauses[0][0]} (${sortedCauses[0][1]})` : '-';
            if(corrEl) corrEl.innerText = sortedCorrections.length > 0 ? `${sortedCorrections[0][0]} (${sortedCorrections[0][1]})` : '-';

            renderPieChart('chartBySection', Object.keys(sectionCounts), Object.values(sectionCounts), chartInstSection, (chart) => chartInstSection = chart, ['#3b82f6', '#f59e0b', '#10b981', '#8b5cf6', '#ef4444', '#64748b']);
            renderPieChart('chartByMachine', Object.keys(machineCounts), Object.values(machineCounts), chartInstMachine, (chart) => chartInstMachine = chart, ['#0ea5e9', '#ec4899', '#14b8a6', '#f43f5e']);
            renderInteractiveBarChart('chartByTestItem', sortedTestItems.map(i => i[0]), sortedTestItems.map(i => i[1]), 'ความถี่ที่พบ', '#8b5cf6', chartInstTestItem, (chart) => chartInstTestItem = chart);
            renderHorizontalBarChart('chartByRootCause', sortedCauses.map(i => i[0]), sortedCauses.map(i => i[1]), 'จำนวน', '#f59e0b', chartInstRootCause, (chart) => chartInstRootCause = chart);
            renderHorizontalBarChart('chartByCorrection', sortedCorrections.map(i => i[0]), sortedCorrections.map(i => i[1]), 'จำนวน', '#10b981', chartInstCorrection, (chart) => chartInstCorrection = chart);
        }

        function renderInteractiveBarChart(canvasId, labels, dataVals, labelName, color, instanceVar, setInstance) {
            const canvas = document.getElementById(canvasId);
            if (!canvas) return;
            if (instanceVar) instanceVar.destroy();
            const ctx = canvas.getContext('2d');
            const newChart = new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: labels.length ? labels : ['ไม่มีข้อมูล'],
                    datasets: [{
                        label: labelName,
                        data: dataVals.length ? dataVals : [0],
                        backgroundColor: color,
                        borderRadius: 6,
                        hoverBackgroundColor: '#6d28d9'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { display: false },
                        tooltip: { callbacks: { footer: () => 'คลิกเพื่อกรองรายการนี้' } }
                    },
                    scales: {
                        y: { beginAtZero: true, ticks: { precision: 0 } },
                        x: { ticks: { autoSkip: false, maxRotation: 45, minRotation: 45 } }
                    },
                    onClick: (e, elements) => {
                        if (elements.length > 0) {
                            const dataIndex = elements[0].index;
                            const clickedLabel = labels[dataIndex];
                            if(clickedLabel !== 'ไม่มีข้อมูล') {
                                const testDropdown = document.getElementById('dashTestItemFilter');
                                let optionExists = Array.from(testDropdown.options).some(opt => opt.value === clickedLabel);

                                    testDropdown.value = clickedLabel;
                                    filterDashboardTable();

                                }
                            }
                        } else {
                            document.getElementById('dashTestItemFilter').value = "";
                            filterDashboardTable();
                        }
                    },
                    onHover: (event, chartElement) => {
                        event.native.target.style.cursor = chartElement[0] ? 'pointer' : 'default';
                    }
                }
            });
            setInstance(newChart);
        }

        function populateDropdownOptions(dropdown, section, preSelectedValue = "") {
            dropdown.innerHTML = "";

            if (!section) {
                const option = document.createElement('option');
                option.value = "";
                option.disabled = true;
                option.selected = true;
                option.textContent = "กรุณาเลือกสาขาก่อน...";
                dropdown.appendChild(option);
                return;
            }

            const tests = testListMap[section] || [];

            if (tests.length === 0) {
                const option = document.createElement('option');
                option.value = "";
                option.disabled = true;
                option.selected = true;
                option.textContent = "ไม่พบรายการตรวจสำหรับสาขานี้";
                dropdown.appendChild(option);
                return;
            }

            const placeholder = document.createElement('option');
            placeholder.value = "";
            placeholder.disabled = true;
            if (!preSelectedValue || !tests.includes(preSelectedValue)) {
                placeholder.selected = true;
            }
            placeholder.textContent = "เลือกรายการตรวจ...";
            dropdown.appendChild(placeholder);

            tests.forEach(test => {
                const option = document.createElement('option');
                option.value = test;
                option.textContent = test;
                if (test === preSelectedValue) {
                    option.selected = true;
                }
                dropdown.appendChild(option);
            });
        }

        function validateEqaItems() {
            const total = parseInt(document.getElementById('eqaTotalItems').value) || 0;
            const passed = parseInt(document.getElementById('eqaPassedItems').value) || 0;
            const alertEl = document.getElementById('eqaErrorAlert');
            const allPassedAlertEl = document.getElementById('eqaAllPassedAlert');
            const childSectionEl = document.getElementById('eqaChildSection');
            const childContainerEl = document.getElementById('eqaChildItemsContainer');
            const submitBtn = document.getElementById('submitEqaBtn');

            if (total > 0 && passed > total) {
                alertEl.classList.remove('hidden');
                allPassedAlertEl.classList.add('hidden');
                childSectionEl.classList.remove('hidden');
                submitBtn.disabled = true;
                submitBtn.classList.add('opacity-50', 'cursor-not-allowed');
                return;
            }

            alertEl.classList.add('hidden');
            submitBtn.disabled = false;
            submitBtn.classList.remove('opacity-50', 'cursor-not-allowed');

            if (total > 0 && passed >= total) {
                // ผ่านครบ 100% (หรือมากกว่า) ไม่ต้องกรอกฟอร์มย่อย Nopass
                childSectionEl.classList.add('hidden');
                allPassedAlertEl.classList.remove('hidden');
                // ล้างการ์ดที่ค้างอยู่ทั้งหมด ป้องกันไม่ให้ input required
                // ที่ถูกซ่อนไว้ไปบล็อกการ submit ฟอร์มแบบเงียบๆ
                if (childContainerEl.children.length > 0) {
                    childContainerEl.innerHTML = '';
                    eqaChildCount = 0;
                }
            } else {
                childSectionEl.classList.remove('hidden');
                allPassedAlertEl.classList.add('hidden');
                // ถ้าไม่มีการ์ดเลย (เพราะเพิ่งล้างไปตอน 100%) ให้เพิ่มกลับมา 1 การ์ด
                if (childContainerEl.children.length === 0) {
                    addEqaChildItem();
                }
            }
        }

        function readEqaFileAsBase64(file) {
            return new Promise((resolve, reject) => {
                if (!file) {
                    resolve(null);
                    return;
                }
                if (file.size > 5 * 1024 * 1024) {
                    reject(new Error('ไฟล์มีขนาดเกิน 5 MB'));
                    return;
                }
                const allowedTypes = ['application/pdf', 'image/jpeg', 'image/png'];
                if (!allowedTypes.includes(file.type)) {
                    reject(new Error('รองรับเฉพาะ PDF, JPG และ PNG'));
                    return;
                }

                const reader = new FileReader();
                reader.onload = () => resolve({
                    name: file.name,
                    mimeType: file.type,
                    base64: String(reader.result).split(',')[1]
                });
                reader.onerror = () => reject(new Error('ไม่สามารถอ่านไฟล์ได้'));
                reader.readAsDataURL(file);
              });
        }
        


        function submitEqaForm(event) {
            event.preventDefault();

            const totalItems = parseInt(document.getElementById('eqaTotalItems').value) || 0;
            const passedItems = parseInt(document.getElementById('eqaPassedItems').value) || 0;
            if (passedItems > totalItems) {
                Swal.fire('ข้อผิดพลาด', 'จำนวน Item ที่ผ่านเกณฑ์ ต้องน้อยกว่าหรือเท่ากับจำนวน Item ทั้งหมด', 'error');
                return;
            }

            const allPassed = (totalItems > 0 && passedItems >= totalItems);

            const childCards = document.querySelectorAll('.eqa-child-item');
            if (!allPassed && childCards.length === 0) {
                Swal.fire('ข้อผิดพลาด', 'กรุณาเพิ่มรายการทดสอบที่ไม่ผ่านเกณฑ์อย่างน้อย 1 รายการ', 'warning');
                return;
            }

            const eqaFormData = {
                type: "EQA",
                parent: {
                    reviewDate: document.getElementById('eqaReviewDate').value,
                    reviewerId: document.getElementById('eqaReviewerId').value,
                    reviewerName: document.getElementById('eqaReviewerName').value,
                    program: document.getElementById('eqaProgram').value,
                    trialNumber: document.getElementById('eqaTrialNumber').value,
                    testDate: document.getElementById('eqaTestDate').value,
                    totalItems: totalItems,
                    passedItems: passedItems,
                    headId: document.getElementById('eqaHeadId').value,
                    headName: document.getElementById('eqaHeadName').value,
                    headAckDate: document.getElementById('eqaHeadAckDate').value
                },
                children: []
            };

            let isChildValid = true;
            if (!allPassed) {
                childCards.forEach(card => {
                    const testRadio = card.querySelector('input[type="radio"][name^="eqaTestNopass_"]:checked');
                    const trendInput = card.querySelector('input[name="eqaTrendHistory[]"]');
                    const correctionInput = card.querySelector('textarea[name="eqaCorrection[]"]');

                    const docProblem = Array.from(card.querySelectorAll('input[type="checkbox"][name^="eqaDocProblem_"]:checked')).map(cb => cb.value).join(', ');
                    const prepProblem = Array.from(card.querySelectorAll('input[type="checkbox"][name^="eqaPrepProblem_"]:checked')).map(cb => cb.value).join(', ');
                    const iqcReview = Array.from(card.querySelectorAll('input[type="checkbox"][name^="eqaIqcReview_"]:checked')).map(cb => cb.value).join(', ');

                    const trendMatch = card.querySelector('input[type="radio"][name^="eqaTrendMatch_"]:checked');
                    const dilution = card.querySelector('input[type="radio"][name^="eqaDilution_"]:checked');
                    const probAmount = card.querySelector('input[type="radio"][name^="eqaProbAmount_"]:checked');
                    const impact = card.querySelector('input[type="radio"][name^="eqaImpact_"]:checked');
                    const conclusion = card.querySelector('input[type="radio"][name^="eqaConclusion_"]:checked');

                    if (!testRadio || !probAmount || !conclusion) {
                        isChildValid = false;
                    }

                    eqaFormData.children.push({
                        testNopass: testRadio ? testRadio.value : '',
                        trendHistory: trendInput ? trendInput.value : '',
                        docProblem: docProblem,
                        prepProblem: prepProblem,
                        iqcReview: iqcReview,
                        trendMatch: trendMatch ? trendMatch.value : '',
                        dilution: dilution ? dilution.value : '',
                        probAmount: probAmount ? probAmount.value : '',
                        impact: impact ? impact.value : '',
                        conclusion: conclusion ? conclusion.value : '',
                        correction: correctionInput ? correctionInput.value : ''
                    });
                });
            }

            if (!isChildValid) {
                Swal.fire('ข้อมูลไม่ครบถ้วน', 'กรุณาระบุข้อมูลรายการที่ไม่ผ่านให้ครบถ้วน โดยเฉพาะ Test, จำนวนตัวอย่าง และ สรุปปัญหา', 'warning');
                return;
            }

            Swal.fire({
                title: 'กำลังบันทึกข้อมูล...',
                html: 'กรุณารอระบบจัดเก็บข้อมูลการทบทวน EQA',
                allowOutsideClick: false,
                didOpen: () => {
                    Swal.showLoading()
                }
            });

            if (typeof google !== 'undefined' && google.script && google.script.run) {
                const selectedFile = document.getElementById('eqaFile').files[0];

                readEqaFileAsBase64(selectedFile)
                    .then(fileObj => {
                        if (!fileObj) {
                            saveEqaFormToSheet(eqaFormData);
                            return;
                        }

                        google.script.run
                            .withSuccessHandler(uploadResponse => {
                                if (!uploadResponse || uploadResponse.status !== 'success') {
                                    onEqaFailure(uploadResponse ? uploadResponse.message : 'อัปโหลดไฟล์ไม่สำเร็จ');
                                    return;
                                }
                                eqaFormData.attachment = {
                                    fileId: uploadResponse.fileId,
                                    fileName: uploadResponse.fileName,
                                    fileUrl: uploadResponse.fileUrl
                                };
                                saveEqaFormToSheet(eqaFormData);
                            })
                            .withFailureHandler(onEqaFailure)
                            .uploadEqaFile(fileObj);
                    })
                    .catch(error => onEqaFailure(error.message || error));
            } else {
                console.log('Mocking EQA Submission Object: ', eqaFormData);
                setTimeout(() => {
                    onEqaSuccess({status: 'success', message: 'บันทึกเรียบร้อยแล้วใน Preview', parentId: 'EQA-MOCK-SUCCESS'});
                }, 1000);
            }

        }






        function onEqaSuccess(response) {
            if(response.status === 'success') {
                const wasEdit = !!eqaEditParentId;
                Swal.fire({
                    icon: 'success',
                    title: wasEdit ? 'แก้ไขข้อมูล EQA สำเร็จ!' : 'บันทึกข้อมูล EQA สำเร็จ!',
                    html: `<div class="text-base">รหัสอ้างอิง: <b>${esc(response.parentId)}</b></div><div class="mt-3 text-sm text-gray-500">สามารถสร้างรายงาน PDF ของรายการนี้ได้ทันที</div>`,
                    showDenyButton: true,
                    showCancelButton: true,
                    confirmButtonText: '📄 สร้าง PDF',
                    denyButtonText: '✏️ แก้ไขต่อ',
                    cancelButtonText: 'ปิด',
                    confirmButtonColor: '#2563eb',
                    denyButtonColor: '#f59e0b'
                }).then((result) => {
                    if (result.isConfirmed) {
                        generateEqaPdfFromClient(response.parentId, wasEdit);
                        return;
                    }
                    if (result.isDenied) {
                        loadEqaForEdit(response.parentId);
                        return;
                    }
                    resetEqaAfterPdf(false, response.parentId);
                });
            } else {
                onEqaFailure(response.message);
            }
        }

        function generateEqaPdfFromClient(parentId, returnToForm) {
            Swal.fire({
                title: 'กำลังสร้าง PDF...',
                html: 'ระบบกำลังจัดรูปแบบรายงาน EQA กรุณารอสักครู่',
                allowOutsideClick: false,
                didOpen: () => Swal.showLoading()
            });
            google.script.run
                .withSuccessHandler(function(res) {
                    if (res && res.status === 'success') {
                        Swal.fire({
                            icon: 'success',
                            title: 'สร้าง PDF สำเร็จ',
                            html: `<div class="text-sm">${esc(res.fileName || 'รายงาน EQA.pdf')}</div>`,
                            showCancelButton: true,
                            confirmButtonText: '📄 เปิด PDF',
                            cancelButtonText: 'ปิด',
                            confirmButtonColor: '#2563eb'
                        }).then(function(r) {
                            if (r.isConfirmed && res.fileUrl) window.open(res.fileUrl, '_blank');
                            resetEqaAfterPdf(returnToForm, parentId);
                        });
                    } else {
                        Swal.fire('สร้าง PDF ไม่สำเร็จ', res ? res.message : 'ไม่ทราบสาเหตุ', 'error');
                    }
                })
                .withFailureHandler(function(err) {
                    Swal.fire('สร้าง PDF ไม่สำเร็จ', String(err), 'error');
                })
                .generateEqaPdf(parentId);
        }

        function resetEqaAfterPdf(returnToForm, parentId) {
            if (returnToForm) {
                loadEqaForEdit(parentId);
                return;
            }
            eqaEditParentId = null; eqaEditAttachment = null; setEqaEditMode(false);
            document.getElementById('eqaForm').reset();
            document.getElementById('eqaReviewDate').valueAsDate = new Date();
            document.getElementById('eqaTestDate').valueAsDate = new Date();
            document.getElementById('eqaHeadAckDate').valueAsDate = new Date();
            document.getElementById('eqaChildItemsContainer').innerHTML = '';
            eqaChildCount = 0;
            document.getElementById('eqaChildSection').classList.remove('hidden');
            document.getElementById('eqaAllPassedAlert').classList.add('hidden');
            document.getElementById('eqaErrorAlert').classList.add('hidden');
            ['eqaReviewerName', 'eqaHeadName'].forEach(id => {
                const el = document.getElementById(id);
                el.setAttribute('readonly', 'true');
                el.className = "w-full p-2.5 border border-gray-200 bg-gray-50 text-gray-500 rounded-lg outline-none transition font-medium";
            });
            document.getElementById('eqaReviewerStatus').innerHTML = "";
            document.getElementById('eqaHeadStatus').innerHTML = "";
            addEqaChildItem();
        }

        function submitForm(event) {

            });

        // Dashboard สถานะ IQC: เรียก getIQCStatus() จาก Code.gs และอัปเดตทุก 60 วินาที
        function loadIqcStatusDashboard() {
            const cards = document.getElementById('iqcStatusCards');
            const icon = document.getElementById('iqcStatusRefreshIcon');
            if (!cards) return;
            if (icon) icon.classList.add('animate-spin');
            cards.innerHTML = '<div class="col-span-full bg-white rounded-xl p-6 text-center text-gray-500"><i class="fa-solid fa-spinner animate-spin mr-2"></i>กำลังโหลดสถานะ...</div>';
            if (!(typeof google !== 'undefined' && google.script && google.script.run)) {
                cards.innerHTML = '<div class="col-span-full bg-red-50 text-red-700 rounded-xl p-5">Dashboard นี้ต้องเปิดผ่าน Google Apps Script Web App และต้องมีฟังก์ชัน getIQCStatus() ใน Code.gs</div>';
                if (icon) icon.classList.remove('animate-spin');
                return;
            }
            var settled = false;
            var timeoutId = setTimeout(function() {
                if (settled) return;
                settled = true;
                if (icon) icon.classList.remove('animate-spin');
                cards.innerHTML = '<div class="col-span-full bg-amber-50 text-amber-800 rounded-xl p-5">เซิร์ฟเวอร์ไม่ตอบกลับภายใน 30 วินาที: ตรวจว่ามีฟังก์ชัน getIQCStatus() ใน Code.gs และได้บันทึก/เผยแพร่ Web App เวอร์ชันล่าสุดแล้ว</div>';
            }, 30000);
            google.script.run.withSuccessHandler(function(items) {
                if (settled) return;
                settled = true; clearTimeout(timeoutId);
                if (icon) icon.classList.remove('animate-spin');
                items = Array.isArray(items) ? items : [];
                const done = items.filter(x => x.done === true).length;
                document.getElementById('iqcStatusDoneCount').textContent = done + ' / ' + items.length;
                document.getElementById('iqcStatusPendingCount').textContent = (items.length - done) + ' / ' + items.length;
                cards.innerHTML = items.map(function(item) {
                    const ok = item.done === true;
                    const title = item.missing ? (item.message || 'ไม่พบแท็บ') : 'คลิกเพื่อเปิดแท็บและกรอกข้อมูล';

                    const cardColor = ok
                        ? 'bg-green-100 border-green-300 border-l-green-500'
                        : 'bg-red-100 border-red-300 border-l-red-500';
                    const nameColor = ok ? 'text-green-900' : 'text-red-900';

                    return '<a href="' + esc(item.url || '#') + '" target="_blank" rel="noopener noreferrer" title="' + esc(title) + '" class="block rounded-xl p-5 shadow-sm border border-l-4 ' + cardColor + ' hover:shadow-md transition">' +
                        '<div class="font-semibold ' + nameColor + ' mb-3">' + esc(item.name || '') + '</div>' +
                        '<div class="text-sm font-bold ' + (ok ? 'text-green-700' : 'text-red-700') + '"><i class="fa-solid ' + (ok ? 'fa-circle-check' : 'fa-circle-xmark') + ' mr-2"></i>' + (ok ? 'ทำแล้ว' : 'ยังไม่ทำ') + '</div></a>';
                }).join('');
                document.getElementById('iqcStatusUpdatedAt').textContent = 'อัปเดตล่าสุด: ' + new Date().toLocaleString('th-TH');
            }).withFailureHandler(function(error) {
                if (settled) return;
                settled = true; clearTimeout(timeoutId);
                if (icon) icon.classList.remove('animate-spin');
                cards.innerHTML = '<div class="col-span-full bg-red-50 text-red-700 rounded-xl p-5">โหลดสถานะไม่สำเร็จ: ' + esc(error && error.message ? error.message : error) + '</div>';
            }).getIQCStatus();
        }

        function loadEqaPassRateSummary() {
          const startDate = document.getElementById('eqaSummaryStartDate').value;
          const endDate = document.getElementById('eqaSummaryEndDate').value;

          google.script.run
            .withSuccessHandler(renderEqaPassRateSummary)
            .withFailureHandler(function(error) {
              console.error('โหลดสรุป EQA ไม่สำเร็จ:', error);
              alert('ไม่สามารถโหลดสรุปผล EQA ได้');
            })
            .getEqaPassRateSummary(startDate, endDate);
        }

        function renderEqaPassRateSummary(res) {
          if (!res || res.status !== 'success') {
            alert(res ? res.message : 'ไม่พบข้อมูลสรุป EQA');
            return;
          }

          document.getElementById('eqaOverallPassRate').textContent =
            res.overall.passRate.toFixed(2) + '%';
          document.getElementById('eqaOverallPassedItems').textContent =
            res.overall.passedItems;
          document.getElementById('eqaOverallTotalItems').textContent =
            res.overall.totalItems;

          const tbody = document.getElementById('eqaPassRateTableBody');
          tbody.innerHTML = '';

          if (!res.byProgram.length) {
            tbody.innerHTML = `
              <tr>
                <td colspan="4" class="text-center text-gray-500 py-6">
                  ไม่พบข้อมูลในช่วงเวลาที่เลือก
                </td>
              </tr>`;
          } else {
            res.byProgram.forEach(function(item) {
              tbody.innerHTML += `
                <tr class="border-b border-gray-100">
                  <td class="px-4 py-3 text-blue-700 font-medium">${item.program}</td>
                  <td class="px-4 py-3 text-center">${item.passedItems}</td>
                  <td class="px-4 py-3 text-center">${item.totalItems}</td>
                  <td class="px-4 py-3 text-center font-bold text-green-600">
                    ${item.passRate.toFixed(2)}%
                  </td>
                </tr>`;
            });
          }

          const ctx = document.getElementById('eqaPassRateChart').getContext('2d');

          if (eqaPassRateChartInstance) {
            eqaPassRateChartInstance.destroy();
          }

          eqaPassRateChartInstance = new Chart(ctx, {
            type: 'bar',
            data: {
              labels: res.byProgram.map(item => item.program),
              datasets: [{
                label: 'ร้อยละผ่านเกณฑ์',
                data: res.byProgram.map(item => item.passRate),
                backgroundColor: '#2563eb',
                borderRadius: 6
              }]
            },
            options: {
              responsive: true,
              maintainAspectRatio: false,
              scales: {
                y: {
                  beginAtZero: true,
                  max: 100,
                  ticks: {
                    callback: value => value + '%'
                  }
                }
              },
              plugins: {
                tooltip: {
                  callbacks: {
                    label: context => context.raw.toFixed(2) + '%'
                  }
                }
              }
            }
          });
        }
